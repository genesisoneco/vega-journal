Pillow>=10

# Vega draws each cover as a bespoke SVG; cairosvg rasterizes it to a PNG for
# social / OG share cards (SVG is not a valid OG image). On Linux it needs the
# system Cairo library:
#   sudo apt-get install -y libcairo2
# If cairosvg is unavailable the SVG still shows on the site and the PNG social
# card falls back to the bundled Pillow template, so the pipeline never breaks.
cairosvg>=2.7
